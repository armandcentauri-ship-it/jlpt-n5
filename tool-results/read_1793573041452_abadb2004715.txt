    53→function shuffle<T>(arr: T[]): T[] {
    54→  const a = [...arr];
    55→  for (let i = a.length - 1; i > 0; i--) {
    56→    const j = Math.floor(Math.random() * (i + 1));
    57→    [a[i], a[j]] = [a[j], a[i]];
    58→  }
    59→  return a;
    60→}

    68→export function Conjugation({ autoStart }: { autoStart?: boolean }) {
    69→  const { words, progressMap } = useAppState();
    70→  const { submit } = useReview();
    71→
    72→  const verbs = useMemo(() => buildVerbList(words), [words]);
    73→
    74→  const [mode, setMode] = useState<Mode>("train");
    75→  const [phase, setPhase] = useState<Phase>("setup");
    76→  const [level, setLevel] = useState<LevelFilter>("N5");
    77→  const [group, setGroup] = useState<GroupFilter>("all");
    78→  const [forms, setForms] = useState<Set<FormKey>>(() => new Set(FORM_KEYS));
    79→  const [size, setSize] = useState(10);
    80→
    81→  const [queue, setQueue] = useState<ConjQuestion[]>([]);
    82→  const [index, setIndex] = useState(0);
    83→  const [input, setInput] = useState("");
    84→  const [status, setStatus] = useState<Status>("answering");
    85→  const [tally, setTally] = useState({ correct: 0, total: 0 });
    86→
    87→  /* ---------------- pool selon les filtres ---------------- */
    88→  const pool = useMemo(
    89→    () => verbs.filter((v) => (level === "all" || v.level === level) && inGroupFilter(v, group)),
    90→    [verbs, level, group]
    91→  );
    92→
    93→  const groupCounts = useMemo(() => {
    94→    const base = verbs.filter((v) => level === "all" || v.level === level);
    95→    return {
    96→      all: base.length,
    97→      godan: base.filter((v) => v.group === "godan" && !v.isSpecialIku && !v.isSpecialAru).length,
    98→      ichidan: base.filter((v) => v.group === "ichidan").length,
    99→      irregular: base.filter((v) => inGroupFilter(v, "irregular")).length,
   100→    };
   101→  }, [verbs, level]);
   102→
   103→  /** Paires (verbe, forme) dont la révision SRS est échue, dans les filtres courants. */
   104→  const duePairs = useMemo(() => {
   105→    const now = Date.now();
   106→    const out: ConjQuestion[] = [];
   107→    const poolIds = new Set(pool.map((v) => v.id));
   108→    for (const p of progressMap.values()) {
   109→      if (p.itemType !== "conj" || !p.itemId.includes(":")) continue;
   110→      const sep = p.itemId.lastIndexOf(":");
   111→      const wordId = p.itemId.slice(0, sep);
   112→      const form = p.itemId.slice(sep + 1) as FormKey;
   113→      if (!poolIds.has(wordId) || !forms.has(form)) continue;
   114→      if (new Date(p.dueAt).getTime() <= now) {
   115→        const verb = pool.find((v) => v.id === wordId);
   116→        if (verb) out.push({ verb, form });
   117→      }
   118→    }
   119→    return out;
   120→  }, [progressMap, pool, forms]);
   121→
   122→  const mastery = useMemo(() => {
   123→    let total = 0;
   124→    let mastered = 0;
   125→    const sel = [...forms];
   126→    for (const v of pool) {
   127→      for (const f of sel) {
   128→        total++;
   129→        const p = progressMap.get(progressKey("conj", `${v.id}:${f}`));
   130→        if (p && p.intervalDays >= MASTERED_DAYS) mastered++;
   131→      }
   132→    }
   133→    return { total, mastered, pct: total ? mastered / total : 0 };
   134→  }, [pool, forms, progressMap]);
   135→
   136→  /* ---------------- session ---------------- */
   137→  const buildQueue = () => {
   138→    const selected = CONJ_FORMS.filter((f) => forms.has(f.key)).map((f) => f.key);
   139→    if (!pool.length || !selected.length) return;
   140→    const maxN = Math.min(size, pool.length * selected.length);
   141→    const seen = new Set<string>();
   142→    const q: ConjQuestion[] = [];
   143→
   144→    // d'abord les révisions échues (SRS), puis du neuf / aléatoire
   145→    for (const d of shuffle(duePairs)) {
   146→      if (q.length >= maxN) break;
   147→      const k = `${d.verb.id}:${d.form}`;
   148→      if (!seen.has(k)) {
   149→        seen.add(k);
   150→        q.push(d);
   151→      }
   152→    }
   153→    if (q.length < maxN) {
   154→      const sv = shuffle(pool);
   155→      const sf = shuffle(selected);
   156→      for (let i = 0; q.length < maxN && i < maxN * 8; i++) {
   157→        const verb = sv[i % sv.length];
   158→        const form = sf[(i + Math.floor(i / Math.max(1, sf.length))) % sf.length];
   159→        const k = `${verb.id}:${form}`;
   160→        if (!seen.has(k)) {
   161→          seen.add(k);
   162→          q.push({ verb, form });
   163→        }
   164→      }
   165→    }
   166→    if (!q.length) return;
   167→    setQueue(shuffle(q));
   168→    setIndex(0);
   169→    setInput("");
   170→    setStatus("answering");
   171→    setTally({ correct: 0, total: 0 });
   172→    setPhase("session");
   173→    setMode("train");
   174→  };
   175→
   176→  /* Lancement direct depuis le tableau de bord (panneau conjugaison) */
   177→  const started = useRef(false);
   178→  useEffect(() => {
   179→    if (autoStart && !started.current) {
   180→      started.current = true;
   181→      buildQueue();
   182→    }
   183→  });
   184→
   185→  const finishItem = (wasCorrect: boolean) => {
   186→    const q = queue[index];
   187→    if (!q) return;
   188→    submit("conj", `${q.verb.id}:${q.form}`, wasCorrect ? 2 : 0, "conj");
   189→    setTally((t) => ({ correct: t.correct + (wasCorrect ? 1 : 0), total: t.total + 1 }));
   190→    if (index + 1 >= queue.length) {
   191→      setPhase("done");
   192→    } else {
   193→      setIndex((i) => i + 1);
   194→      setInput("");
   195→      setStatus("answering");
   196→    }
   197→  };
   198→
   199→  const check = () => {
   200→    if (!input.trim()) return;
   201→    const q = queue[index];
   202→    if (!q) return;
   203→    const f = conjugateVerb(q.verb)[q.form];
   204→    /* finalise le « n » en attente du clavier kana */
   205→    const { result } = checkConj(finalizeKana(input), conjRefs(f));
   206→    setStatus(result === "correct" ? "correct" : result === "close" ? "close" : "wrong");
   207→  };
   208→
   209→  const reveal = () => setStatus("revealed");
   210→
   211→  const toggleForm = (key: FormKey) => {
   212→    setForms((prev) => {
   213→      const n = new Set(prev);
   214→      if (n.has(key)) n.delete(key);
   215→      else n.add(key);
   216→      return n;
   217→    });
   218→  };
   219→
   220→  /* ================= TABLEAUX ================= */
   221→  const [search, setSearch] = useState("");
   222→  const [tLevel, setTLevel] = useState<LevelFilter>("N5");
   223→  const [tGroup, setTGroup] = useState<GroupFilter>("all");
   224→  const [selectedVerb, setSelectedVerb] = useState<VerbItem | null>(null);
   225→  const [hideAnswers, setHideAnswers] = useState(false);
   226→  const [revealedRows, setRevealedRows] = useState<Set<string>>(new Set());
   227→
   228→  const tPool = useMemo(() => {
   229→    const q = search.trim().toLowerCase();
   230→    return verbs
   231→      .filter((v) => (tLevel === "all" || v.level === tLevel) && inGroupFilter(v, tGroup))
   232→      .filter(
   233→        (v) =>
   234→          !q ||
   235→          v.word.toLowerCase().includes(q) ||
   236→          v.reading.includes(q) ||
   237→          v.meaningFr.toLowerCase().includes(q)
   238→      );
   239→  }, [verbs, tLevel, tGroup, search]);
   240→
   241→  const verbForms = useMemo(
   242→    () => (selectedVerb ? conjugateVerb(selectedVerb) : null),
   243→    [selectedVerb]
   244→  );
   245→
   246→  const conjStat = (verb: VerbItem, form: FormKey) =>
   247→    progressMap.get(progressKey("conj", `${verb.id}:${form}`));
   248→
   249→  /* ---------------- rendu ---------------- */
   250→  return (
   251→    <div className="mx-auto w-full max-w-2xl space-y-6 px-4 pb-10 pt-6">
   252→      <SectionTitle kanji="動詞">Conjugaison</SectionTitle>
   253→
   254→      {/* bascule Entraînement / Tableaux */}
   255→      <div className="flex gap-2">
   256→        <button
   257→          type="button"
   258→          onClick={() => setMode("train")}
   259→          aria-pressed={mode === "train"}
   260→          className={cn(
   261→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   262→            mode === "train" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   263→          )}
   264→        >
   265→          <GraduationCap className="h-4 w-4" aria-hidden /> Entraînement
   266→        </button>
   267→        <button
   268→          type="button"
   269→          onClick={() => setMode("tables")}
   270→          aria-pressed={mode === "tables"}
   271→          className={cn(
   272→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   273→            mode === "tables" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   274→          )}
   275→        >
   276→          <BookOpen className="h-4 w-4" aria-hidden /> Tableaux
   277→        </button>
   278→      </div>
   279→
   280→      {mode === "train" ? (
   281→        phase === "setup" ? (
   282→          <TrainSetup
   283→            pool={pool}
   284→            groupCounts={groupCounts}
   285→            level={level}
   286→            setLevel={setLevel}
   287→            group={group}
   288→            setGroup={setGroup}
   289→            forms={forms}
   290→            toggleForm={toggleForm}
   291→            setForms={setForms}
   292→            size={size}
   293→            setSize={setSize}
   294→            mastery={mastery}
   295→            nDue={duePairs.length}
   296→            onStart={buildQueue}
   297→          />
   298→        ) : phase === "done" ? (
   299→          <TrainDone
   300→            tally={tally}
   301→            onAgain={buildQueue}
   302→            onSettings={() => setPhase("setup")}
   303→            onTables={() => {
   304→              setMode("tables");
   305→              setPhase("setup");
   306→            }}
   307→          />
   308→        ) : (
   309→          <TrainSession
   310→            queue={queue}
   311→            index={index}
   312→            input={input}
   313→            setInput={setInput}
   314→            status={status}
   315→            check={check}
   316→            reveal={reveal}
   317→            finishItem={finishItem}
   318→            quit={() => setPhase("setup")}
   319→          />
   320→        )
   321→      ) : (
   322→        <Tables
   323→          verbs={tPool}
   324→          total={verbs.length}
   325→          search={search}
   326→          setSearch={setSearch}
   327→          tLevel={tLevel}
   328→          setTLevel={setTLevel}
   329→          tGroup={tGroup}
   330→          setTGroup={setTGroup}
   331→          selectedVerb={selectedVerb}
   332→          setSelectedVerb={(v) => {
   333→            setSelectedVerb(v);
   334→            setRevealedRows(new Set());
   335→          }}
   336→          hideAnswers={hideAnswers}
   337→          setHideAnswers={(b) => {
   338→            setHideAnswers(b);
   339→            setRevealedRows(new Set());
   340→          }}
   341→          revealedRows={revealedRows}
   342→          toggleRow={(k) =>
   343→            setRevealedRows((prev) => {
   344→              const n = new Set(prev);
   345→              if (n.has(k)) n.delete(k);
   346→              else n.add(k);
   347→              return n;
   348→            })
   349→          }
   350→          verbForms={verbForms}
   351→          conjStat={conjStat}
   352→        />
   353→      )}
   354→    </div>
   355→  );
   356→}

   360→function TrainSetup(props: {
   361→  pool: VerbItem[];
   362→  groupCounts: { all: number; godan: number; ichidan: number; irregular: number };
   363→  level: LevelFilter;
   364→  setLevel: (l: LevelFilter) => void;
   365→  group: GroupFilter;
   366→  setGroup: (g: GroupFilter) => void;

   398→        <div>

   402→              Tous ({groupCounts.all})
   403→            </FilterChip>

   420→              <Button
   421→                type="button"
   422→                variant="ghost"
   423→                size="sm"
   424→                className="h-7 px-2 text-xs text-muted-foreground"
   425→                onClick={() => setForms(new Set(FORM_KEYS))}
   426→              >
   427→                Tout
   428→              </Button>
   429→              <Button
   430→                type="button"
   431→                variant="ghost"
   432→                size="sm"
   433→                className="h-7 px-2 text-xs text-muted-foreground"

   435→              >
   436→                Bases
   437→              </Button>
   438→            </div>
   439→          </div>
   440→          <div className="grid gap-1.5 sm:grid-cols-2">
   441→            {CONJ_FORMS.map((f) => {
   442→              const on = forms.has(f.key);
   443→              return (
   444→                <button

   446→                  type="button"
   447→                  onClick={() => toggleForm(f.key)}
   448→                  aria-pressed={on}
   449→                  className={cn(
   450→                    "flex items-center justify-between rounded-lg border px-3 py-2 text-left text-sm transition-all",
   451→                    on ? "border-primary/50 bg-primary/5" : "bg-card text-muted-foreground"
   452→                  )}
   453→                >
   454→                  <span>
   455→                    {f.label}
   456→                    <span className="ml-1.5 font-jp text-xs text-muted-foreground">{f.hint}</span>
   457→                  </span>
   458→                  <span
   459→                    className={cn(
   460→                      "ml-2 h-4 w-4 shrink-0 rounded-full border",
   461→                      on ? "border-primary bg-primary" : "border-border"
   462→                    )}
   463→                    aria-hidden
   464→                  />
   465→                </button>
   466→              );
   467→            })}
   468→          </div>
   469→        </div>
   470→
   471→        <div>
   472→          <p className="mb-2 text-sm font-medium">Nombre de questions</p>
   473→          <div className="flex gap-2">
   474→            {[5, 10, 15, 20].map((n) => (
   475→              <Button
   476→                key={n}
   477→                type="button"
   478→                size="sm"
   479→                variant={size === n ? "default" : "outline"}
   480→                onClick={() => setSize(n)}
   481→                className="rounded-full"
   482→              >
   483→                {n}
   484→              </Button>
   485→            ))}
   486→          </div>
   487→        </div>
   488→
   489→        {mastery.total > 0 && (
   490→          <div>
   491→            <div className="mb-1 flex items-center justify-between text-xs text-muted-foreground">
   492→              <span>Maîtrisées</span>
   493→              <span>
   494→                {mastery.mastered}/{mastery.total}

   496→            </div>
   497→            <div className="h-1.5 overflow-hidden rounded-full bg-border">

   504→        )}
   505→
   506→        <Button onClick={onStart} disabled={!pool.length || !forms.size} className="h-11 w-full rounded-full">
   507→          Commencer {nPairs > 0 && <span className="ml-1 text-xs opacity-80">({nPairs} combinaisons)</span>}
   508→          <ArrowRight className="ml-2 h-4 w-4" aria-hidden />
   509→        </Button>
   510→      </CardContent>
   511→    </Card>
   512→  );
   513→}

   515→function TrainSession(props: {
   516→  queue: ConjQuestion[];
   517→  index: number;
   518→  input: string;
   519→  setInput: (v: string) => void;
   520→  status: Status;
   521→  check: () => void;
   522→  reveal: () => void;
   523→  finishItem: (wasCorrect: boolean) => void;
   524→  quit: () => void;
   525→}) {
   526→  const { queue, index, input, setInput, status, check, reveal, finishItem, quit } = props;
   527→  const q = queue[index];
   528→  if (!q) return null;
   529→  const { verb, form } = q;
   530→  const formDef = CONJ_FORMS.find((f) => f.key === form)!;
   531→  const forms = conjugateVerb(verb);
   532→  const answer = forms[form];
   533→  const answered = status !== "answering";
   534→  const showInput = status === "answering";
   535→
   536→  return (
   537→    <div className="space-y-4">
   538→      <div className="flex items-center gap-3">
   539→        <div className="h-2 flex-1 overflow-hidden rounded-full bg-border">
   540→          <div
   541→            className="h-full rounded-full bg-primary transition-all duration-300"
   542→            style={{ width: `${(index / queue.length) * 100}%` }}
   543→          />
   544→        </div>
   545→        <span className="text-xs text-muted-foreground">
   546→          {index + 1}/{queue.length}
   547→        </span>
   548→        <Button variant="ghost" size="sm" onClick={quit} className="text-muted-foreground">
   549→          Quitter
   550→        </Button>
   551→      </div>
   552→
   553→      <Card className="anim-fade-up">
   554→        <CardContent className="space-y-4 p-6">
   555→          <div className="flex items-center justify-between">
   556→            <span className="inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-xs font-semibold text-primary">
   557→              {formDef.label}
