    52→function shuffle<T>(arr: T[]): T[] {
    53→  const a = [...arr];
    54→  for (let i = a.length - 1; i > 0; i--) {
    55→    const j = Math.floor(Math.random() * (i + 1));
    56→    [a[i], a[j]] = [a[j], a[i]];
    57→  }
    58→  return a;
    59→}

    75→  const [level, setLevel] = useState<LevelFilter>("N5");

    77→  const [forms, setForms] = useState<Set<FormKey>>(() => new Set(FORM_KEYS));
    78→  const [size, setSize] = useState(10);
    79→
    80→  const [queue, setQueue] = useState<ConjQuestion[]>([]);
    81→  const [index, setIndex] = useState(0);
    82→  const [input, setInput] = useState("");
    83→  const [status, setStatus] = useState<Status>("answering");
    84→  const [tally, setTally] = useState({ correct: 0, total: 0 });
    85→
    86→  /* ---------------- pool selon les filtres ---------------- */
    87→  const pool = useMemo(
    88→    () => verbs.filter((v) => (level === "all" || v.level === level) && inGroupFilter(v, group)),
    89→    [verbs, level, group]
    90→  );
    91→
    92→  const groupCounts = useMemo(() => {
    93→    const base = verbs.filter((v) => level === "all" || v.level === level);
    94→    return {
    95→      all: base.length,
    96→      godan: base.filter((v) => v.group === "godan" && !v.isSpecialIku && !v.isSpecialAru).length,
    97→      ichidan: base.filter((v) => v.group === "ichidan").length,
    98→      irregular: base.filter((v) => inGroupFilter(v, "irregular")).length,
    99→    };
   100→  }, [verbs, level]);
   101→
   102→  /** Paires (verbe, forme) dont la révision SRS est échue, dans les filtres courants. */
   103→  const duePairs = useMemo(() => {
   104→    const now = Date.now();
   105→    const out: ConjQuestion[] = [];
   106→    const poolIds = new Set(pool.map((v) => v.id));
   107→    for (const p of progressMap.values()) {
   108→      if (p.itemType !== "conj" || !p.itemId.includes(":")) continue;
   109→      const sep = p.itemId.lastIndexOf(":");
   110→      const wordId = p.itemId.slice(0, sep);
   111→      const form = p.itemId.slice(sep + 1) as FormKey;
   112→      if (!poolIds.has(wordId) || !forms.has(form)) continue;
   113→      if (new Date(p.dueAt).getTime() <= now) {
   114→        const verb = pool.find((v) => v.id === wordId);
   115→        if (verb) out.push({ verb, form });
   116→      }
   117→    }
   118→    return out;
   119→  }, [progressMap, pool, forms]);
   120→
   121→  const mastery = useMemo(() => {
   122→    let total = 0;
   123→    let mastered = 0;
   124→    const sel = [...forms];
   125→    for (const v of pool) {
   126→      for (const f of sel) {
   127→        total++;
   128→        const p = progressMap.get(progressKey("conj", `${v.id}:${f}`));
   129→        if (p && p.intervalDays >= MASTERED_DAYS) mastered++;
   130→      }
   131→    }
   132→    return { total, mastered, pct: total ? mastered / total : 0 };
   133→  }, [pool, forms, progressMap]);
   134→
   135→  /* ---------------- session ---------------- */
   136→  const buildQueue = () => {
   137→    const selected = CONJ_FORMS.filter((f) => forms.has(f.key)).map((f) => f.key);
   138→    if (!pool.length || !selected.length) return;
   139→    const maxN = Math.min(size, pool.length * selected.length);
   140→    const seen = new Set<string>();
   141→    const q: ConjQuestion[] = [];
   142→
   143→    // d'abord les révisions échues (SRS), puis du neuf / aléatoire
   144→    for (const d of shuffle(duePairs)) {
   145→      if (q.length >= maxN) break;
   146→      const k = `${d.verb.id}:${d.form}`;
   147→      if (!seen.has(k)) {
   148→        seen.add(k);
   149→        q.push(d);
   150→      }
   151→    }
   152→    if (q.length < maxN) {
   153→      const sv = shuffle(pool);
   154→      const sf = shuffle(selected);
   155→      for (let i = 0; q.length < maxN && i < maxN * 8; i++) {
   156→        const verb = sv[i % sv.length];
   157→        const form = sf[(i + Math.floor(i / Math.max(1, sf.length))) % sf.length];
   158→        const k = `${verb.id}:${form}`;
   159→        if (!seen.has(k)) {
   160→          seen.add(k);
   161→          q.push({ verb, form });
   162→        }
   163→      }
   164→    }
   165→    if (!q.length) return;
   166→    setQueue(shuffle(q));
   167→    setIndex(0);
   168→    setInput("");
   169→    setStatus("answering");
   170→    setTally({ correct: 0, total: 0 });
   171→    setPhase("session");
   172→    setMode("train");
   173→  };
   174→
   175→  const finishItem = (wasCorrect: boolean) => {
   176→    const q = queue[index];
   177→    if (!q) return;
   178→    submit("conj", `${q.verb.id}:${q.form}`, wasCorrect ? 2 : 0, "conj");
   179→    setTally((t) => ({ correct: t.correct + (wasCorrect ? 1 : 0), total: t.total + 1 }));
   180→    if (index + 1 >= queue.length) {
   181→      setPhase("done");
   182→    } else {
   183→      setIndex((i) => i + 1);
   184→      setInput("");
   185→      setStatus("answering");
   186→    }
   187→  };
   188→
   189→  const check = () => {
   190→    if (!input.trim()) return;
   191→    const q = queue[index];
   192→    if (!q) return;
   193→    const f = conjugateVerb(q.verb)[q.form];
   194→    const { result } = checkConj(input, conjRefs(f));
   195→    setStatus(result === "correct" ? "correct" : result === "close" ? "close" : "wrong");
   196→  };
   197→
   198→  const reveal = () => setStatus("revealed");
   199→
   200→  const toggleForm = (key: FormKey) => {
   201→    setForms((prev) => {
   202→      const n = new Set(prev);
   203→      if (n.has(key)) n.delete(key);
   204→      else n.add(key);
   205→      return n;
   206→    });
   207→  };
   208→
   209→  /* ================= TABLEAUX ================= */
   210→  const [search, setSearch] = useState("");
   211→  const [tLevel, setTLevel] = useState<LevelFilter>("N5");
   212→  const [tGroup, setTGroup] = useState<GroupFilter>("all");
   213→  const [selectedVerb, setSelectedVerb] = useState<VerbItem | null>(null);
   214→  const [hideAnswers, setHideAnswers] = useState(false);
   215→  const [revealedRows, setRevealedRows] = useState<Set<string>>(new Set());
   216→
   217→  const tPool = useMemo(() => {
   218→    const q = search.trim().toLowerCase();
   219→    return verbs
   220→      .filter((v) => (tLevel === "all" || v.level === tLevel) && inGroupFilter(v, tGroup))
   221→      .filter(
   222→        (v) =>
   223→          !q ||
   224→          v.word.toLowerCase().includes(q) ||
   225→          v.reading.includes(q) ||
   226→          v.meaningFr.toLowerCase().includes(q)
   227→      );
   228→  }, [verbs, tLevel, tGroup, search]);
   229→
   230→  const verbForms = useMemo(
   231→    () => (selectedVerb ? conjugateVerb(selectedVerb) : null),
   232→    [selectedVerb]
   233→  );
   234→
   235→  const conjStat = (verb: VerbItem, form: FormKey) =>
   236→    progressMap.get(progressKey("conj", `${verb.id}:${form}`));
   237→
   238→  /* ---------------- rendu ---------------- */
   239→  return (
   240→    <div className="mx-auto w-full max-w-2xl space-y-6 px-4 pb-10 pt-6">
   241→      <SectionTitle kanji="動詞">Conjugaison</SectionTitle>
   242→
   243→      {/* bascule Entraînement / Tableaux */}
   244→      <div className="flex gap-2">
   245→        <button
   246→          type="button"
   247→          onClick={() => setMode("train")}
   248→          aria-pressed={mode === "train"}
   249→          className={cn(
   250→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   251→            mode === "train" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   252→          )}
   253→        >
   254→          <GraduationCap className="h-4 w-4" aria-hidden /> Entraînement
   255→        </button>
   256→        <button
   257→          type="button"
   258→          onClick={() => setMode("tables")}
   259→          aria-pressed={mode === "tables"}
   260→          className={cn(
   261→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   262→            mode === "tables" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   263→          )}
   264→        >
   265→          <BookOpen className="h-4 w-4" aria-hidden /> Tableaux

   268→
   269→      {mode === "train" ? (

   273→            groupCounts={groupCounts}
   274→            level={level}

   276→            group={group}
   277→            setGroup={setGroup}
   278→            forms={forms}
   279→            toggleForm={toggleForm}
   280→            setForms={setForms}
   281→            size={size}
   282→            setSize={setSize}
   283→            mastery={mastery}
   284→            nDue={duePairs.length}
   285→            onStart={buildQueue}
   286→          />
   287→        ) : phase === "done" ? (
   288→          <TrainDone
   289→            tally={tally}
   290→            onAgain={buildQueue}
   291→            onSettings={() => setPhase("setup")}
   292→            onTables={() => {
   293→              setMode("tables");
   294→              setPhase("setup");
   295→            }}
   296→          />
   297→        ) : (
   298→          <TrainSession
   299→            queue={queue}
   300→            index={index}
   301→            input={input}
   302→            setInput={setInput}
   303→            status={status}
   304→            check={check}
   305→            reveal={reveal}
   306→            finishItem={finishItem}
   307→            quit={() => setPhase("setup")}
   308→          />
   309→        )
   310→      ) : (
   311→        <Tables
   312→          verbs={tPool}
   313→          total={verbs.length}
   314→          search={search}
   315→          setSearch={setSearch}
   316→          tLevel={tLevel}
   317→          setTLevel={setTLevel}
   318→          tGroup={tGroup}
   319→          setTGroup={setTGroup}
   320→          selectedVerb={selectedVerb}
   321→          setSelectedVerb={(v) => {
   322→            setSelectedVerb(v);
   323→            setRevealedRows(new Set());
   324→          }}
   325→          hideAnswers={hideAnswers}
   326→          setHideAnswers={(b) => {
   327→            setHideAnswers(b);
   328→            setRevealedRows(new Set());
   329→          }}
   330→          revealedRows={revealedRows}
   331→          toggleRow={(k) =>
   332→            setRevealedRows((prev) => {
   333→              const n = new Set(prev);
   334→              if (n.has(k)) n.delete(k);
   335→              else n.add(k);
   336→              return n;
   337→            })
   338→          }
   339→          verbForms={verbForms}
   340→          conjStat={conjStat}
   341→        />
   342→      )}
   343→    </div>
   344→  );
   345→}

   512→function TrainSession(props: {
   513→  queue: ConjQuestion[];
   514→  index: number;
   515→  input: string;
   516→  setInput: (v: string) => void;
   517→  status: Status;
   518→  check: () => void;
   519→  reveal: () => void;
   520→  finishItem: (wasCorrect: boolean) => void;
   521→  quit: () => void;
   522→}) {
   523→  const { queue, index, input, setInput, status, check, reveal, finishItem, quit } = props;
   524→  const q = queue[index];
   525→  if (!q) return null;
   526→  const { verb, form } = q;
   527→  const formDef = CONJ_FORMS.find((f) => f.key === form)!;
   528→  const forms = conjugateVerb(verb);
   529→  const answer = forms[form];
   530→  const answered = status !== "answering";
   531→  const showInput = status === "answering";
   532→
   533→  return (
   534→    <div className="space-y-4">
   535→      <div className="flex items-center gap-3">
   536→        <div className="h-2 flex-1 overflow-hidden rounded-full bg-border">
   537→          <div
   538→            className="h-full rounded-full bg-primary transition-all duration-300"
   539→            style={{ width: `${(index / queue.length) * 100}%` }}
   540→          />
   541→        </div>
   542→        <span className="text-xs text-muted-foreground">
   543→          {index + 1}/{queue.length}
   544→        </span>
   545→        <Button variant="ghost" size="sm" onClick={quit} className="text-muted-foreground">
   546→          Quitter
   547→        </Button>
   548→      </div>
   549→
   550→      <Card className="anim-fade-up">
