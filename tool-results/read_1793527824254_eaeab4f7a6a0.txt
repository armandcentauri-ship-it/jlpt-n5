    52→    52→function shuffle<T>(arr: T[]): T[] {
    53→    53→  const a = [...arr];
    54→    54→  for (let i = a.length - 1; i > 0; i--) {
    55→    55→    const j = Math.floor(Math.random() * (i + 1));
    56→    56→    [a[i], a[j]] = [a[j], a[i]];
    57→    57→  }
    58→    58→  return a;
    59→    59→}

    67→    67→export function Conjugation() {
    68→    68→  const { words, progressMap } = useAppState();

    70→    70→
    71→    71→  const verbs = useMemo(() => buildVerbList(words), [words]);
    72→    72→

    75→    75→  const [level, setLevel] = useState<LevelFilter>("N5");
    76→    76→  const [group, setGroup] = useState<GroupFilter>("all");
    77→    77→  const [forms, setForms] = useState<Set<FormKey>>(() => new Set(FORM_KEYS));
    78→    78→  const [size, setSize] = useState(10);
    79→    79→
    80→    80→  const [queue, setQueue] = useState<ConjQuestion[]>([]);
    81→    81→  const [index, setIndex] = useState(0);
    82→    82→  const [input, setInput] = useState("");
    83→    83→  const [status, setStatus] = useState<Status>("answering");
    84→    84→  const [tally, setTally] = useState({ correct: 0, total: 0 });
    85→    85→
    86→    86→  /* ---------------- pool selon les filtres ---------------- */
    87→    87→  const pool = useMemo(
    88→    88→    () => verbs.filter((v) => (level === "all" || v.level === level) && inGroupFilter(v, group)),
    89→    89→    [verbs, level, group]
    90→    90→  );
    91→    91→
    92→    92→  const groupCounts = useMemo(() => {
    93→    93→    const base = verbs.filter((v) => level === "all" || v.level === level);
    94→    94→    return {
    95→    95→      all: base.length,
    96→    96→      godan: base.filter((v) => v.group === "godan" && !v.isSpecialIku && !v.isSpecialAru).length,
    97→    97→      ichidan: base.filter((v) => v.group === "ichidan").length,
    98→    98→      irregular: base.filter((v) => inGroupFilter(v, "irregular")).length,
    99→    99→    };
   100→   100→  }, [verbs, level]);
   101→   101→
   102→   102→  /** Paires (verbe, forme) dont la révision SRS est échue, dans les filtres courants. */
   103→   103→  const duePairs = useMemo(() => {
   104→   104→    const now = Date.now();
   105→   105→    const out: ConjQuestion[] = [];
   106→   106→    const poolIds = new Set(pool.map((v) => v.id));
   107→   107→    for (const p of progressMap.values()) {
   108→   108→      if (p.itemType !== "conj" || !p.itemId.includes(":")) continue;
   109→   109→      const sep = p.itemId.lastIndexOf(":");
   110→   110→      const wordId = p.itemId.slice(0, sep);
   111→   111→      const form = p.itemId.slice(sep + 1) as FormKey;
   112→   112→      if (!poolIds.has(wordId) || !forms.has(form)) continue;
   113→   113→      if (new Date(p.dueAt).getTime() <= now) {
   114→   114→        const verb = pool.find((v) => v.id === wordId);
   115→   115→        if (verb) out.push({ verb, form });
   116→   116→      }
   117→   117→    }
   118→   118→    return out;
   119→   119→  }, [progressMap, pool, forms]);
   120→   120→
   121→   121→  const mastery = useMemo(() => {
   122→   122→    let total = 0;
   123→   123→    let mastered = 0;
   124→   124→    const sel = [...forms];
   125→   125→    for (const v of pool) {
   126→   126→      for (const f of sel) {
   127→   127→        total++;
   128→   128→        const p = progressMap.get(progressKey("conj", `${v.id}:${f}`));
   129→   129→        if (p && p.intervalDays >= MASTERED_DAYS) mastered++;
   130→   130→      }
   131→   131→    }
   132→   132→    return { total, mastered, pct: total ? mastered / total : 0 };
   133→   133→  }, [pool, forms, progressMap]);

   137→   137→    const selected = CONJ_FORMS.filter((f) => forms.has(f.key)).map((f) => f.key);
   138→   138→    if (!pool.length || !selected.length) return;
   139→   139→    const maxN = Math.min(size, pool.length * selected.length);
   140→   140→    const seen = new Set<string>();
   141→   141→    const q: ConjQuestion[] = [];
   142→   142→
   143→   143→    // d'abord les révisions échues (SRS), puis du neuf / aléatoire
   144→   144→    for (const d of shuffle(duePairs)) {
   145→   145→      if (q.length >= maxN) break;
   146→   146→      const k = `${d.verb.id}:${d.form}`;
   147→   147→      if (!seen.has(k)) {
   148→   148→        seen.add(k);
   149→   149→        q.push(d);
   150→   150→      }
   151→   151→    }
   152→   152→    if (q.length < maxN) {
   153→   153→      const sv = shuffle(pool);
   154→   154→      const sf = shuffle(selected);
   155→   155→      for (let i = 0; q.length < maxN && i < maxN * 8; i++) {
   156→   156→        const verb = sv[i % sv.length];
   157→   157→        const form = sf[(i + Math.floor(i / Math.max(1, sf.length))) % sf.length];
   158→   158→        const k = `${verb.id}:${form}`;
   159→   159→        if (!seen.has(k)) {
   160→   160→          seen.add(k);
   161→   161→          q.push({ verb, form });
   162→   162→        }
   163→   163→      }
   164→   164→    }
   165→   165→    if (!q.length) return;
   166→   166→    setQueue(shuffle(q));
   167→   167→    setIndex(0);
   168→   168→    setInput("");
   169→   169→    setStatus("answering");
   170→   170→    setTally({ correct: 0, total: 0 });
   171→   171→    setPhase("session");
   172→   172→    setMode("train");
   173→   173→  };
   174→   174→
   175→   175→  const finishItem = (wasCorrect: boolean) => {
   176→   176→    const q = queue[index];
   177→   177→    if (!q) return;
   178→   178→    submit("conj", `${q.verb.id}:${q.form}`, wasCorrect ? 2 : 0, "conj");
   179→   179→    setTally((t) => ({ correct: t.correct + (wasCorrect ? 1 : 0), total: t.total + 1 }));
   180→   180→    if (index + 1 >= queue.length) {
   181→   181→      setPhase("done");
   182→   182→    } else {
   183→   183→      setIndex((i) => i + 1);
   184→   184→      setInput("");
   185→   185→      setStatus("answering");
   186→   186→    }
   187→   187→  };

   196→   196→  };

   198→   198→  const reveal = () => setStatus("revealed");
   199→   199→
   200→   200→  const toggleForm = (key: FormKey) => {
   201→   201→    setForms((prev) => {
   202→   202→      const n = new Set(prev);
   203→   203→      if (n.has(key)) n.delete(key);
   204→   204→      else n.add(key);
   205→   205→      return n;
   206→   206→    });
   207→   207→  };
   208→   208→
   209→   209→  /* ================= TABLEAUX ================= */
   210→   210→  const [search, setSearch] = useState("");

   222→   222→        (v) =>

   226→   226→          v.meaningFr.toLowerCase().includes(q)

   228→   228→  }, [verbs, tLevel, tGroup, search]);
   229→   229→
   230→   230→  const verbForms = useMemo(

   232→   232→    [selectedVerb]
   233→   233→  );

   235→   235→  const conjStat = (verb: VerbItem, form: FormKey) =>
   236→   236→    progressMap.get(progressKey("conj", `${verb.id}:${form}`));
   237→   237→
   238→   238→  /* ---------------- rendu ---------------- */

   245→   245→        <button
   246→   246→          type="button"
   247→   247→          onClick={() => setMode("train")}
   248→   248→          aria-pressed={mode === "train"}
   249→   249→          className={cn(
   250→   250→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   251→   251→            mode === "train" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   252→   252→          )}

   254→   254→          <GraduationCap className="h-4 w-4" aria-hidden /> Entraînement

   256→   256→        <button
   257→   257→          type="button"
   258→   258→          onClick={() => setMode("tables")}
   259→   259→          aria-pressed={mode === "tables"}
   260→   260→          className={cn(
   261→   261→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   262→   262→            mode === "tables" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   263→   263→          )}
   264→   264→        >
   265→   265→          <BookOpen className="h-4 w-4" aria-hidden /> Tableaux
   266→   266→        </button>
   267→   267→      </div>
   268→   268→
   269→   269→      {mode === "train" ? (
   270→   270→        phase === "setup" ? (
   271→   271→          <TrainSetup
   272→   272→            pool={pool}
   273→   273→            groupCounts={groupCounts}
   274→   274→            level={level}
   275→   275→            setLevel={setLevel}
   276→   276→            group={group}
   277→   277→            setGroup={setGroup}
   278→   278→            forms={forms}
   279→   279→            toggleForm={toggleForm}
   280→   280→            setForms={setForms}
   281→   281→            size={size}
   282→   282→            setSize={setSize}
   283→   283→            mastery={mastery}
   284→   284→            nDue={duePairs.length}
   285→   285→            onStart={buildQueue}
   286→   286→          />
   287→   287→        ) : phase === "done" ? (
   288→   288→          <TrainDone
   289→   289→            tally={tally}
   290→   290→            onAgain={buildQueue}
   291→   291→            onSettings={() => setPhase("setup")}
   292→   292→            onTables={() => {
   293→   293→              setMode("tables");
   294→   294→              setPhase("setup");
   295→   295→            }}
   296→   296→          />
   297→   297→        ) : (
   298→   298→          <TrainSession
   299→   299→            queue={queue}
   300→   300→            index={index}
   301→   301→            input={input}
   302→   302→            setInput={setInput}
   303→   303→            status={status}
   304→   304→            check={check}
   305→   305→            reveal={reveal}
   306→   306→            finishItem={finishItem}
   307→   307→            quit={() => setPhase("setup")}
   308→   308→          />
   309→   309→        )
   310→   310→      ) : (
   311→   311→        <Tables
   312→   312→          verbs={tPool}
   313→   313→          total={verbs.length}
   314→   314→          search={search}
   315→   315→          setSearch={setSearch}
   316→   316→          tLevel={tLevel}
   317→   317→          setTLevel={setTLevel}
   318→   318→          tGroup={tGroup}
   319→   319→          setTGroup={setTGroup}
   320→   320→          selectedVerb={selectedVerb}
   321→   321→          setSelectedVerb={(v) => {
   322→   322→            setSelectedVerb(v);
   323→   323→            setRevealedRows(new Set());
   324→   324→          }}
   325→   325→          hideAnswers={hideAnswers}
   326→   326→          setHideAnswers={(b) => {
   327→   327→            setHideAnswers(b);
   328→   328→            setRevealedRows(new Set());
   329→   329→          }}
   330→   330→          revealedRows={revealedRows}
   331→   331→          toggleRow={(k) =>
   332→   332→            setRevealedRows((prev) => {
   333→   333→              const n = new Set(prev);
   334→   334→              if (n.has(k)) n.delete(k);
   335→   335→              else n.add(k);
   336→   336→              return n;
   337→   337→            })
   338→   338→          }
   339→   339→          verbForms={verbForms}
   340→   340→          conjStat={conjStat}
   341→   341→        />
   342→   342→      )}
   343→   343→    </div>
   344→   344→  );
   345→   345→}
