    53→    53→function shuffle<T>(arr: T[]): T[] {
    54→    54→  const a = [...arr];
    55→    55→  for (let i = a.length - 1; i > 0; i--) {
    56→    56→    const j = Math.floor(Math.random() * (i + 1));
    57→    57→    [a[i], a[j]] = [a[j], a[i]];
    58→    58→  }
    59→    59→  return a;
    60→    60→}

    68→    68→export function Conjugation({ autoStart }: { autoStart?: boolean }) {
    69→    69→  const { words, progressMap } = useAppState();
    70→    70→  const { submit } = useReview();
    71→    71→
    72→    72→  const verbs = useMemo(() => buildVerbList(words), [words]);
    73→    73→
    74→    74→  const [mode, setMode] = useState<Mode>("train");
    75→    75→  const [phase, setPhase] = useState<Phase>("setup");
    76→    76→  const [level, setLevel] = useState<LevelFilter>("N5");
    77→    77→  const [group, setGroup] = useState<GroupFilter>("all");
    78→    78→  const [forms, setForms] = useState<Set<FormKey>>(() => new Set(FORM_KEYS));
    79→    79→  const [size, setSize] = useState(10);
    80→    80→
    81→    81→  const [queue, setQueue] = useState<ConjQuestion[]>([]);
    82→    82→  const [index, setIndex] = useState(0);
    83→    83→  const [input, setInput] = useState("");
    84→    84→  const [status, setStatus] = useState<Status>("answering");
    85→    85→  const [tally, setTally] = useState({ correct: 0, total: 0 });
    86→    86→
    87→    87→  /* ---------------- pool selon les filtres ---------------- */
    88→    88→  const pool = useMemo(
    89→    89→    () => verbs.filter((v) => (level === "all" || v.level === level) && inGroupFilter(v, group)),
    90→    90→    [verbs, level, group]
    91→    91→  );
    92→    92→
    93→    93→  const groupCounts = useMemo(() => {
    94→    94→    const base = verbs.filter((v) => level === "all" || v.level === level);
    95→    95→    return {
    96→    96→      all: base.length,
    97→    97→      godan: base.filter((v) => v.group === "godan" && !v.isSpecialIku && !v.isSpecialAru).length,
    98→    98→      ichidan: base.filter((v) => v.group === "ichidan").length,
    99→    99→      irregular: base.filter((v) => inGroupFilter(v, "irregular")).length,
   100→   100→    };
   101→   101→  }, [verbs, level]);
   102→   102→
   103→   103→  /** Paires (verbe, forme) dont la révision SRS est échue, dans les filtres courants. */
   104→   104→  const duePairs = useMemo(() => {
   105→   105→    const now = Date.now();
   106→   106→    const out: ConjQuestion[] = [];
   107→   107→    const poolIds = new Set(pool.map((v) => v.id));
   108→   108→    for (const p of progressMap.values()) {
   109→   109→      if (p.itemType !== "conj" || !p.itemId.includes(":")) continue;
   110→   110→      const sep = p.itemId.lastIndexOf(":");
   111→   111→      const wordId = p.itemId.slice(0, sep);
   112→   112→      const form = p.itemId.slice(sep + 1) as FormKey;
   113→   113→      if (!poolIds.has(wordId) || !forms.has(form)) continue;
   114→   114→      if (new Date(p.dueAt).getTime() <= now) {
   115→   115→        const verb = pool.find((v) => v.id === wordId);
   116→   116→        if (verb) out.push({ verb, form });
   117→   117→      }
   118→   118→    }
   119→   119→    return out;
   120→   120→  }, [progressMap, pool, forms]);
   121→   121→
   122→   122→  const mastery = useMemo(() => {
   123→   123→    let total = 0;
   124→   124→    let mastered = 0;
   125→   125→    const sel = [...forms];
   126→   126→    for (const v of pool) {
   127→   127→      for (const f of sel) {
   128→   128→        total++;
   129→   129→        const p = progressMap.get(progressKey("conj", `${v.id}:${f}`));
   130→   130→        if (p && p.intervalDays >= MASTERED_DAYS) mastered++;
   131→   131→      }
   132→   132→    }
   133→   133→    return { total, mastered, pct: total ? mastered / total : 0 };
   134→   134→  }, [pool, forms, progressMap]);
   135→   135→
   136→   136→  /* ---------------- session ---------------- */
   137→   137→  const buildQueue = () => {
   138→   138→    const selected = CONJ_FORMS.filter((f) => forms.has(f.key)).map((f) => f.key);
   139→   139→    if (!pool.length || !selected.length) return;
   140→   140→    const maxN = Math.min(size, pool.length * selected.length);
   141→   141→    const seen = new Set<string>();
   142→   142→    const q: ConjQuestion[] = [];
   143→   143→
   144→   144→    // d'abord les révisions échues (SRS), puis du neuf / aléatoire
   145→   145→    for (const d of shuffle(duePairs)) {
   146→   146→      if (q.length >= maxN) break;
   147→   147→      const k = `${d.verb.id}:${d.form}`;
   148→   148→      if (!seen.has(k)) {
   149→   149→        seen.add(k);
   150→   150→        q.push(d);
   151→   151→      }
   152→   152→    }
   153→   153→    if (q.length < maxN) {
   154→   154→      const sv = shuffle(pool);
   155→   155→      const sf = shuffle(selected);
   156→   156→      for (let i = 0; q.length < maxN && i < maxN * 8; i++) {
   157→   157→        const verb = sv[i % sv.length];
   158→   158→        const form = sf[(i + Math.floor(i / Math.max(1, sf.length))) % sf.length];
   159→   159→        const k = `${verb.id}:${form}`;
   160→   160→        if (!seen.has(k)) {
   161→   161→          seen.add(k);
   162→   162→          q.push({ verb, form });
   163→   163→        }
   164→   164→      }
   165→   165→    }
   166→   166→    if (!q.length) return;
   167→   167→    setQueue(shuffle(q));
   168→   168→    setIndex(0);
   169→   169→    setInput("");
   170→   170→    setStatus("answering");
   171→   171→    setTally({ correct: 0, total: 0 });
   172→   172→    setPhase("session");
   173→   173→    setMode("train");
   174→   174→  };
   175→   175→
   176→   176→  /* Lancement direct depuis le tableau de bord (panneau conjugaison) */
   177→   177→  const started = useRef(false);
   178→   178→  useEffect(() => {
   179→   179→    if (autoStart && !started.current) {
   180→   180→      started.current = true;
   181→   181→      buildQueue();
   182→   182→    }
   183→   183→  });
   184→   184→
   185→   185→  const finishItem = (wasCorrect: boolean) => {
   186→   186→    const q = queue[index];
   187→   187→    if (!q) return;
   188→   188→    submit("conj", `${q.verb.id}:${q.form}`, wasCorrect ? 2 : 0, "conj");
   189→   189→    setTally((t) => ({ correct: t.correct + (wasCorrect ? 1 : 0), total: t.total + 1 }));
   190→   190→    if (index + 1 >= queue.length) {
   191→   191→      setPhase("done");
   192→   192→    } else {
   193→   193→      setIndex((i) => i + 1);
   194→   194→      setInput("");
   195→   195→      setStatus("answering");
   196→   196→    }
   197→   197→  };
   198→   198→
   199→   199→  const check = () => {
   200→   200→    if (!input.trim()) return;
   201→   201→    const q = queue[index];
   202→   202→    if (!q) return;
   203→   203→    const f = conjugateVerb(q.verb)[q.form];
   204→   204→    /* finalise le « n » en attente du clavier kana */
   205→   205→    const { result } = checkConj(finalizeKana(input), conjRefs(f));
   206→   206→    setStatus(result === "correct" ? "correct" : result === "close" ? "close" : "wrong");
   207→   207→  };
   208→   208→
   209→   209→  const reveal = () => setStatus("revealed");
   210→   210→
   211→   211→  const toggleForm = (key: FormKey) => {
   212→   212→    setForms((prev) => {
   213→   213→      const n = new Set(prev);
   214→   214→      if (n.has(key)) n.delete(key);
   215→   215→      else n.add(key);
   216→   216→      return n;
   217→   217→    });
   218→   218→  };
   219→   219→
   220→   220→  /* ================= TABLEAUX ================= */
   221→   221→  const [search, setSearch] = useState("");
   222→   222→  const [tLevel, setTLevel] = useState<LevelFilter>("N5");
   223→   223→  const [tGroup, setTGroup] = useState<GroupFilter>("all");
   224→   224→  const [selectedVerb, setSelectedVerb] = useState<VerbItem | null>(null);
   225→   225→  const [hideAnswers, setHideAnswers] = useState(false);
   226→   226→  const [revealedRows, setRevealedRows] = useState<Set<string>>(new Set());
   227→   227→
   228→   228→  const tPool = useMemo(() => {
   229→   229→    const q = search.trim().toLowerCase();
   230→   230→    return verbs
   231→   231→      .filter((v) => (tLevel === "all" || v.level === tLevel) && inGroupFilter(v, tGroup))
   232→   232→      .filter(
   233→   233→        (v) =>
   234→   234→          !q ||
   235→   235→          v.word.toLowerCase().includes(q) ||
   236→   236→          v.reading.includes(q) ||
   237→   237→          v.meaningFr.toLowerCase().includes(q)
   238→   238→      );
   239→   239→  }, [verbs, tLevel, tGroup, search]);
   240→   240→
   241→   241→  const verbForms = useMemo(
   242→   242→    () => (selectedVerb ? conjugateVerb(selectedVerb) : null),
   243→   243→    [selectedVerb]
   244→   244→  );
   245→   245→
   246→   246→  const conjStat = (verb: VerbItem, form: FormKey) =>
   247→   247→    progressMap.get(progressKey("conj", `${verb.id}:${form}`));
   248→   248→
   249→   249→  /* ---------------- rendu ---------------- */
   250→   250→  return (
   251→   251→    <div className="mx-auto w-full max-w-2xl space-y-6 px-4 pb-10 pt-6">
   252→   252→      <SectionTitle kanji="動詞">Conjugaison</SectionTitle>
   253→   253→
   254→   254→      {/* bascule Entraînement / Tableaux */}
   255→   255→      <div className="flex gap-2">
   256→   256→        <button
   257→   257→          type="button"
   258→   258→          onClick={() => setMode("train")}
   259→   259→          aria-pressed={mode === "train"}
   260→   260→          className={cn(
   261→   261→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   262→   262→            mode === "train" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   263→   263→          )}
   264→   264→        >
   265→   265→          <GraduationCap className="h-4 w-4" aria-hidden /> Entraînement
   266→   266→        </button>
   267→   267→        <button
   268→   268→          type="button"
   269→   269→          onClick={() => setMode("tables")}
   270→   270→          aria-pressed={mode === "tables"}
   271→   271→          className={cn(
   272→   272→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   273→   273→            mode === "tables" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   274→   274→          )}
   275→   275→        >
   276→   276→          <BookOpen className="h-4 w-4" aria-hidden /> Tableaux
   277→   277→        </button>
   278→   278→      </div>
   279→   279→
   280→   280→      {mode === "train" ? (
   281→   281→        phase === "setup" ? (
   282→   282→          <TrainSetup
   283→   283→            pool={pool}
   284→   284→            groupCounts={groupCounts}
   285→   285→            level={level}
   286→   286→            setLevel={setLevel}
   287→   287→            group={group}
   288→   288→            setGroup={setGroup}
   289→   289→            forms={forms}
   290→   290→            toggleForm={toggleForm}
   291→   291→            setForms={setForms}
   292→   292→            size={size}
   293→   293→            setSize={setSize}
   294→   294→            mastery={mastery}
   295→   295→            nDue={duePairs.length}
   296→   296→            onStart={buildQueue}
   297→   297→          />
   298→   298→        ) : phase === "done" ? (
   299→   299→          <TrainDone
   300→   300→            tally={tally}
   301→   301→            onAgain={buildQueue}
   302→   302→            onSettings={() => setPhase("setup")}
   303→   303→            onTables={() => {
   304→   304→              setMode("tables");
   305→   305→              setPhase("setup");
   306→   306→            }}
   307→   307→          />
   308→   308→        ) : (
   309→   309→          <TrainSession
   310→   310→            queue={queue}
   311→   311→            index={index}
   312→   312→            input={input}
   313→   313→            setInput={setInput}
   314→   314→            status={status}
   315→   315→            check={check}
   316→   316→            reveal={reveal}
   317→   317→            finishItem={finishItem}
   318→   318→            quit={() => setPhase("setup")}
   319→   319→          />
   320→   320→        )
   321→   321→      ) : (
   322→   322→        <Tables
   323→   323→          verbs={tPool}
   324→   324→          total={verbs.length}
   325→   325→          search={search}
   326→   326→          setSearch={setSearch}
   327→   327→          tLevel={tLevel}
   328→   328→          setTLevel={setTLevel}
   329→   329→          tGroup={tGroup}
   330→   330→          setTGroup={setTGroup}
   331→   331→          selectedVerb={selectedVerb}
   332→   332→          setSelectedVerb={(v) => {
   333→   333→            setSelectedVerb(v);
   334→   334→            setRevealedRows(new Set());
   335→   335→          }}
   336→   336→          hideAnswers={hideAnswers}
   337→   337→          setHideAnswers={(b) => {
   338→   338→            setHideAnswers(b);
   339→   339→            setRevealedRows(new Set());
   340→   340→          }}
   341→   341→          revealedRows={revealedRows}
   342→   342→          toggleRow={(k) =>
   343→   343→            setRevealedRows((prev) => {
   344→   344→              const n = new Set(prev);
   345→   345→              if (n.has(k)) n.delete(k);
   346→   346→              else n.add(k);
   347→   347→              return n;
   348→   348→            })
   349→   349→          }
   350→   350→          verbForms={verbForms}
   351→   351→          conjStat={conjStat}
   352→   352→        />
   353→   353→      )}
   354→   354→    </div>
   355→   355→  );
   356→   356→}
