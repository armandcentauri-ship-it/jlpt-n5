    54→function shuffle<T>(arr: T[]): T[] {
    55→  const a = [...arr];
    56→  for (let i = a.length - 1; i > 0; i--) {
    57→    const j = Math.floor(Math.random() * (i + 1));
    58→    [a[i], a[j]] = [a[j], a[i]];
    59→  }
    60→  return a;
    61→}

    69→export function Conjugation({ autoStart }: { autoStart?: boolean }) {
    70→  const { words, progressMap } = useAppState();
    71→  const { submit } = useReview();
    72→
    73→  const verbs = useMemo(() => buildVerbList(words), [words]);
    74→
    75→  const [mode, setMode] = useState<Mode>("train");
    76→  const [phase, setPhase] = useState<Phase>("setup");
    77→  const [level, setLevel] = useState<LevelFilter>("N5");
    78→  const [group, setGroup] = useState<GroupFilter>("all");
    79→  const [forms, setForms] = useState<Set<FormKey>>(() => new Set(N5_FORM_KEYS));
    80→  const [size, setSize] = useState(10);
    81→
    82→  const [queue, setQueue] = useState<ConjQuestion[]>([]);
    83→  const [index, setIndex] = useState(0);
    84→  const [input, setInput] = useState("");
    85→  const [status, setStatus] = useState<Status>("answering");
    86→  const [tally, setTally] = useState({ correct: 0, total: 0 });
    87→
    88→  /* ---------------- pool selon les filtres ---------------- */
    89→  const pool = useMemo(
    90→    () => verbs.filter((v) => (level === "all" || v.level === level) && inGroupFilter(v, group)),
    91→    [verbs, level, group]
    92→  );
    93→
    94→  const groupCounts = useMemo(() => {
    95→    const base = verbs.filter((v) => level === "all" || v.level === level);
    96→    return {
    97→      all: base.length,
    98→      godan: base.filter((v) => v.group === "godan" && !v.isSpecialIku && !v.isSpecialAru).length,
    99→      ichidan: base.filter((v) => v.group === "ichidan").length,
   100→      irregular: base.filter((v) => inGroupFilter(v, "irregular")).length,
   101→    };
   102→  }, [verbs, level]);
   103→
   104→  /** Paires (verbe, forme) dont la révision SRS est échue, dans les filtres courants. */
   105→  const duePairs = useMemo(() => {
   106→    const now = Date.now();
   107→    const out: ConjQuestion[] = [];
   108→    const poolIds = new Set(pool.map((v) => v.id));
   109→    for (const p of progressMap.values()) {
   110→      if (p.itemType !== "conj" || !p.itemId.includes(":")) continue;
   111→      const sep = p.itemId.lastIndexOf(":");
   112→      const wordId = p.itemId.slice(0, sep);
   113→      const form = p.itemId.slice(sep + 1) as FormKey;
   114→      if (!poolIds.has(wordId) || !forms.has(form)) continue;
   115→      if (new Date(p.dueAt).getTime() <= now) {
   116→        const verb = pool.find((v) => v.id === wordId);
   117→        if (verb) out.push({ verb, form });
   118→      }
   119→    }
   120→    return out;
   121→  }, [progressMap, pool, forms]);
   122→
   123→  const mastery = useMemo(() => {
   124→    let total = 0;
   125→    let mastered = 0;
   126→    const sel = [...forms];
   127→    for (const v of pool) {
   128→      for (const f of sel) {
   129→        total++;
   130→        const p = progressMap.get(progressKey("conj", `${v.id}:${f}`));
   131→        if (p && p.intervalDays >= MASTERED_DAYS) mastered++;
   132→      }
   133→    }
   134→    return { total, mastered, pct: total ? mastered / total : 0 };
   135→  }, [pool, forms, progressMap]);
   136→
   137→  /* ---------------- session ---------------- */
   138→  const buildQueue = () => {
   139→    const selected = CONJ_FORMS.filter((f) => forms.has(f.key)).map((f) => f.key);
   140→    if (!pool.length || !selected.length) return;
   141→    const maxN = Math.min(size, pool.length * selected.length);
   142→    const seen = new Set<string>();
   143→    const q: ConjQuestion[] = [];
   144→
   145→    // d'abord les révisions échues (SRS), puis du neuf / aléatoire
   146→    for (const d of shuffle(duePairs)) {
   147→      if (q.length >= maxN) break;
   148→      const k = `${d.verb.id}:${d.form}`;
   149→      if (!seen.has(k)) {
   150→        seen.add(k);
   151→        q.push(d);
   152→      }
   153→    }
   154→    if (q.length < maxN) {
   155→      const sv = shuffle(pool);
   156→      const sf = shuffle(selected);
   157→      for (let i = 0; q.length < maxN && i < maxN * 8; i++) {
   158→        const verb = sv[i % sv.length];
   159→        const form = sf[(i + Math.floor(i / Math.max(1, sf.length))) % sf.length];
   160→        const k = `${verb.id}:${form}`;
   161→        if (!seen.has(k)) {
   162→          seen.add(k);
   163→          q.push({ verb, form });
   164→        }
   165→      }
   166→    }
   167→    if (!q.length) return;
   168→    setQueue(shuffle(q));
   169→    setIndex(0);
   170→    setInput("");
   171→    setStatus("answering");
   172→    setTally({ correct: 0, total: 0 });
   173→    setPhase("session");
   174→    setMode("train");
   175→  };
   176→
   177→  /* Lancement direct depuis le tableau de bord (panneau conjugaison) */
   178→  const started = useRef(false);
   179→  useEffect(() => {
   180→    if (autoStart && !started.current) {
   181→      started.current = true;
   182→      buildQueue();
   183→    }
   184→  });
   185→
   186→  const finishItem = (wasCorrect: boolean) => {
   187→    const q = queue[index];
   188→    if (!q) return;
   189→    submit("conj", `${q.verb.id}:${q.form}`, wasCorrect ? 2 : 0, "conj");
   190→    setTally((t) => ({ correct: t.correct + (wasCorrect ? 1 : 0), total: t.total + 1 }));
   191→    if (index + 1 >= queue.length) {
   192→      setPhase("done");
   193→    } else {
   194→      setIndex((i) => i + 1);
   195→      setInput("");
   196→      setStatus("answering");
   197→    }
   198→  };
   199→
   200→  const check = () => {
   201→    if (!input.trim()) return;
   202→    const q = queue[index];
   203→    if (!q) return;
   204→    const f = conjugateVerb(q.verb)[q.form];
   205→    /* finalise le « n » en attente du clavier kana */
   206→    const { result } = checkConj(finalizeKana(input), conjRefs(f));
   207→    setStatus(result === "correct" ? "correct" : result === "close" ? "close" : "wrong");
   208→  };
   209→
   210→  const reveal = () => setStatus("revealed");
   211→
   212→  const toggleForm = (key: FormKey) => {
   213→    setForms((prev) => {
   214→      const n = new Set(prev);
   215→      if (n.has(key)) n.delete(key);
   216→      else n.add(key);
   217→      return n;
   218→    });
   219→  };
   220→
   221→  /* ================= TABLEAUX ================= */
   222→  const [search, setSearch] = useState("");
   223→  const [tLevel, setTLevel] = useState<LevelFilter>("N5");
   224→  const [tGroup, setTGroup] = useState<GroupFilter>("all");
   225→  const [selectedVerb, setSelectedVerb] = useState<VerbItem | null>(null);
   226→  const [hideAnswers, setHideAnswers] = useState(false);
   227→  const [revealedRows, setRevealedRows] = useState<Set<string>>(new Set());
   228→
   229→  const tPool = useMemo(() => {
   230→    const q = search.trim().toLowerCase();
   231→    return verbs
   232→      .filter((v) => (tLevel === "all" || v.level === tLevel) && inGroupFilter(v, tGroup))
   233→      .filter(
   234→        (v) =>
   235→          !q ||
   236→          v.word.toLowerCase().includes(q) ||
   237→          v.reading.includes(q) ||
   238→          v.meaningFr.toLowerCase().includes(q)
   239→      );
   240→  }, [verbs, tLevel, tGroup, search]);
   241→
   242→  const verbForms = useMemo(
   243→    () => (selectedVerb ? conjugateVerb(selectedVerb) : null),
   244→    [selectedVerb]
   245→  );
   246→
   247→  /* un verbe est toujours sélectionné en mode Tableaux → la table de synthèse est visible d'emblée */
   248→  useEffect(() => {
   249→    if (mode === "tables" && (!selectedVerb || !tPool.some((v) => v.id === selectedVerb.id))) {
   250→      setSelectedVerb(tPool[0] ?? null);
   251→      setRevealedRows(new Set());
   252→    }
   253→  }, [mode, tPool, selectedVerb]);
   254→
   255→  const conjStat = (verb: VerbItem, form: FormKey) =>
   256→    progressMap.get(progressKey("conj", `${verb.id}:${form}`));
   257→
   258→  /* ---------------- rendu ---------------- */
   259→  return (
   260→    <div className="mx-auto w-full max-w-2xl space-y-6 px-4 pb-10 pt-6">
   261→      <SectionTitle kanji="動詞">Conjugaison</SectionTitle>
   262→
   263→      {/* bascule Entraînement / Tableaux */}
   264→      <div className="flex gap-2">
   265→        <button
   266→          type="button"
   267→          onClick={() => setMode("train")}
   268→          aria-pressed={mode === "train"}
   269→          className={cn(
   270→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   271→            mode === "train" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   272→          )}
   273→        >
   274→          <GraduationCap className="h-4 w-4" aria-hidden /> Entraînement
   275→        </button>
   276→        <button
   277→          type="button"
   278→          onClick={() => setMode("tables")}
   279→          aria-pressed={mode === "tables"}
   280→          className={cn(
   281→            "flex flex-1 items-center justify-center gap-2 rounded-xl border p-3 text-sm font-medium transition-all",
   282→            mode === "tables" ? "border-primary bg-primary/5 ring-2 ring-primary/20" : "bg-card hover:border-primary/40"
   283→          )}
   284→        >
   285→          <BookOpen className="h-4 w-4" aria-hidden /> Tableaux
   286→        </button>
   287→      </div>
   288→
   289→      {mode === "train" ? (
   290→        phase === "setup" ? (
   291→          <TrainSetup
   292→            pool={pool}
   293→            groupCounts={groupCounts}
   294→            level={level}
   295→            setLevel={setLevel}
   296→            group={group}
   297→            setGroup={setGroup}
   298→            forms={forms}
   299→            toggleForm={toggleForm}
   300→            setForms={setForms}
   301→            size={size}
   302→            setSize={setSize}
   303→            mastery={mastery}
   304→            nDue={duePairs.length}
   305→            onStart={buildQueue}
   306→          />
   307→        ) : phase === "done" ? (
   308→          <TrainDone
   309→            tally={tally}
   310→            onAgain={buildQueue}
   311→            onSettings={() => setPhase("setup")}
   312→            onTables={() => {
   313→              setMode("tables");
   314→              setPhase("setup");
   315→            }}
   316→          />
   317→        ) : (
   318→          <TrainSession
   319→            queue={queue}
   320→            index={index}
   321→            input={input}
   322→            setInput={setInput}
   323→            status={status}
   324→            check={check}
   325→            reveal={reveal}
   326→            finishItem={finishItem}
   327→            quit={() => setPhase("setup")}
   328→          />
   329→        )
   330→      ) : (
   331→        <Tables
   332→          verbs={tPool}
   333→          total={verbs.length}
   334→          search={search}
   335→          setSearch={setSearch}
   336→          tLevel={tLevel}
   337→          setTLevel={setTLevel}
   338→          tGroup={tGroup}
   339→          setTGroup={setTGroup}
   340→          selectedVerb={selectedVerb}
   341→          setSelectedVerb={(v) => {
   342→            setSelectedVerb(v);
   343→            setRevealedRows(new Set());
   344→          }}
   345→          hideAnswers={hideAnswers}
   346→          setHideAnswers={(b) => {
   347→            setHideAnswers(b);
   348→            setRevealedRows(new Set());
   349→          }}
   350→          revealedRows={revealedRows}
   351→          toggleRow={(k) =>
   352→            setRevealedRows((prev) => {
   353→              const n = new Set(prev);
   354→              if (n.has(k)) n.delete(k);
   355→              else n.add(k);
   356→              return n;
   357→            })
   358→          }
   359→          verbForms={verbForms}
   360→          conjStat={conjStat}
   361→        />
   362→      )}
   363→    </div>
   364→  );
   365→}

   371→  groupCounts: { all: number; godan: number; ichidan: number; irregular: number };

   439→                type="button"
   440→                variant="ghost"

   442→                className="h-7 px-2 text-xs text-muted-foreground"

   445→                Bases

   447→            </div>

   450→            {CONJ_FORMS.map((f) => {
   451→              const on = forms.has(f.key);

   511→            <div className="h-1.5 overflow-hidden rounded-full bg-border">
   512→              <div
   513→                className="h-full rounded-full bg-matcha transition-all duration-500"
   514→                style={{ width: `${Math.min(100, mastery.pct * 100)}%` }}

   516→            </div>
   517→          </div>

   519→
   520→        <Button onClick={onStart} disabled={!pool.length || !forms.size} className="h-11 w-full rounded-full">

   522→          <ArrowRight className="ml-2 h-4 w-4" aria-hidden />
   523→        </Button>
   524→      </CardContent>
   525→    </Card>
   526→  );
   527→}

   529→function TrainSession(props: {
   530→  queue: ConjQuestion[];
   531→  index: number;
   532→  input: string;
   533→  setInput: (v: string) => void;
   534→  status: Status;
   535→  check: () => void;
   536→  reveal: () => void;
   537→  finishItem: (wasCorrect: boolean) => void;
   538→  quit: () => void;
   539→}) {
   540→  const { queue, index, input, setInput, status, check, reveal, finishItem, quit } = props;
   541→  const q = queue[index];
   542→  if (!q) return null;
   543→  const { verb, form } = q;
   544→  const formDef = CONJ_FORMS.find((f) => f.key === form)!;
   545→  const forms = conjugateVerb(verb);
   546→  const answer = forms[form];
   547→  const answered = status !== "answering";
   548→  const showInput = status === "answering";
   549→
   550→  return (
   551→    <div className="space-y-4">
   552→      <div className="flex items-center gap-3">
   553→        <div className="h-2 flex-1 overflow-hidden rounded-full bg-border">
   554→          <div
   555→            className="h-full rounded-full bg-primary transition-all duration-300"
